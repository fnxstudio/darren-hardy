<script>
/* TCEB Founders Event: custom code for the Spiffy checkouts (base, -200-off, -1000-off, -pif).
   One file for everything we add inside Spiffy. The demo pages run this same file, so what the team
   sees in the demo is what gets installed. Hand-over: wrap in script tags and paste into each checkout's
   Settings > custom code. Keep it free of dollar signs: Spiffy's save step rewrites a dollar sign followed by
   digits, so prices are written as \u0024 (the browser still shows a dollar sign). */

/* 0. Wait for Spiffy's checkout to settle. Spiffy's server sends the checkout as finished HTML, then its app takes over
      that HTML in the browser and keeps adding parts of the form for a moment after (even after its own "ready" hook).
      Changing the page during that makes the app crash ("HierarchyRequestError: appendChild"), and on phones the form
      can then never finish loading. So everything below that touches the form waits until the page has fully loaded
      AND Spiffy has stopped changing the page for half a second (6s cap). The CSS in part 1 is safe right away.
      On a page without Spiffy's app (our demo) it runs straight away. */
function tcebAfterMount(fn) {
  var done = false;
  function go() { if (done) return; done = true; setTimeout(fn, 0); }
  if (!document.getElementById('__nuxt')) { go(); return; }
  function settle() {
    var quiet, cap = setTimeout(finish, 6000);
    var mo = new MutationObserver(function () { clearTimeout(quiet); quiet = setTimeout(finish, 500); });
    function finish() { clearTimeout(quiet); clearTimeout(cap); mo.disconnect(); go(); }
    mo.observe(document.documentElement, { childList: true, subtree: true });
    quiet = setTimeout(finish, 500);
  }
  if (document.readyState === 'complete') settle(); else window.addEventListener('load', settle);
}

/* 1. Order summary.
      a. Show the total a second time, right above the Join button, after address, shipping and tax. Spiffy
         already draws this copy (.order-summary-mobile) and shows it on phones; this shows it on desktop too.
      b. Hide the code chip, and Spiffy's own "Discount" row (it sits apart from the items, under a divider).
      c. Add two lines to Order Details, styled like the item lines above them, from Spiffy's own order numbers:
         - the Rearden Sponsorship: Spiffy leaves free (zero-price) items out of Order Details, so it is added
           here as "REARDEN SPONSORSHIP (5,000 dollar value)  INCLUDED" whenever that item is in the order;
         - the credit, when a code is applied: "10 Book Credit (-200)" for 200OFF, "100 Book Credit (-1,000)"
           for 1000OFF, "Credit (-amount)" for any other code (shown with a dollar sign). The monthly -200-off and
           -1000-off carts have the credit built into their price (no code), so they show it as
           "10 Book Credit (-200)  INCLUDED" / "100 Book Credit (-1,000)  INCLUDED". The amount is Spiffy's own discount, so it always
           matches the total. Only the display changes. Redrawn whenever Spiffy redraws the summary. */
(function () {
  var css = document.createElement('style');
  css.textContent = '.order-summary-mobile{display:flex!important}' +
    '.coupon-code,.line--discount{display:none!important}' +
    '.tceb-line .line-item__group_price{white-space:nowrap}' +
    '.tceb-inc{font-size:.8em;letter-spacing:.08em}';
  document.head.appendChild(css);

  var BOOKS = { '200OFF': '10', '1000OFF': '100' };   // code -> books in the pre-print order
  function money(cents) {
    var d = cents / 100;
    return '\u0024' + d.toLocaleString('en-US', { minimumFractionDigits: d % 1 ? 2 : 0, maximumFractionDigits: 2 });
  }
  function lines() {
    var ck = null;
    try { ck = window['\u0024nuxt'].$store.state.checkout; } catch (e) {}
    if (!ck || !ck.summary) return [];
    var out = [], s = ck.summary;
    var rearden = (s.line_items || []).some(function (li) {   // one line, even if the cart holds it twice
      return /rearden/i.test((li.pretty_name || '') + ' ' + (li.name || '')) && !li.total;
    });
    if (rearden) out.push(['REARDEN SPONSORSHIP (\u00245,000 value)', '<span class="tceb-inc">INCLUDED</span>']);
    if (s.discount_amount > 0) {
      var books = BOOKS[String((ck.promo && ck.promo.code) || '').toUpperCase()];
      out.push([books ? books + ' Book Credit' : 'Credit', '(-' + money(s.discount_amount) + ')']);
    } else {
      // Monthly -200-off and -1000-off carts: the credit is already in their price, so name it here
      var built = /-200-off\//.test(location.pathname + '/') ? ['10', 20000] : /-1000-off\//.test(location.pathname + '/') ? ['100', 100000] : null;
      if (built) out.push([built[0] + ' Book Credit (-' + money(built[1]) + ')', '<span class="tceb-inc">INCLUDED</span>']);
    }
    return out;
  }
  function draw() {
    var want = lines(), key = JSON.stringify(want);
    document.querySelectorAll('.order-summary__slide').forEach(function (list) {
      var mine = list.querySelectorAll('.tceb-line');
      if (list.getAttribute('data-tceb') === key && mine.length === want.length &&
          (!mine.length || list.lastElementChild === mine[mine.length - 1])) return;   // already right
      mine.forEach(function (el) { el.remove(); });
      var proto = list.querySelector('.line-item__group');
      want.forEach(function (w) {
        var el = document.createElement('div');
        if (proto) for (var i = 0; i < proto.attributes.length; i++) {
          if (/^data-v-/.test(proto.attributes[i].name)) el.setAttribute(proto.attributes[i].name, '');
        }
        el.className = 'list-group-item row no-gutters line-item__group tceb-line';
        el.innerHTML = '<div class="col"><p class="line-item__group_name">' + w[0] + '</p></div>' +
          '<div class="text-right col-auto"><p class="line-item__group_price">' + w[1] + '</p></div>';
        list.appendChild(el);
      });
      list.setAttribute('data-tceb', key);
    });
  }
  tcebAfterMount(function () {
    draw();
    new MutationObserver(draw).observe(document.documentElement, { childList: true, subtree: true });
  });
})();

/* 2. Spiffy's Quantity block can't go to 0, so the checkout keeps the "Extra Team Seats" dropdown (0 to 10) as the
   real control. This script hides the dropdown and shows a − / + counter that sets it, exactly as if the buyer
   had picked a number. If the script doesn't run, the plain dropdown shows instead and still works. */
(function () {
  var css = document.createElement('style');
  css.textContent =
    '.tceb-team-row{display:flex;justify-content:flex-start;align-items:center;gap:14px}' +
    '.tceb-team-label{font-weight:700;font-size:17px}' +
    '.tceb-stepper{display:inline-flex;border:1px solid #dedede;border-radius:2px;background:#fff}' +
    '.tceb-stepper button{width:42px;height:44px;border:0;background:#fff;color:#000;font-size:20px;line-height:1;cursor:pointer}' +
    '.tceb-stepper button:disabled{color:#c4c4c4;cursor:default}' +
    '.tceb-stepper output{display:grid;place-items:center;width:48px;border-left:1px solid #dedede;border-right:1px solid #dedede;font-size:16px;font-weight:700}';
  document.head.appendChild(css);

  function findSelect() {
    var blocks = document.querySelectorAll('[id^="block-"]');
    for (var i = 0; i < blocks.length; i++) {
      var sel = blocks[i].querySelector('select');
      if (sel && /extra team seats/i.test(blocks[i].textContent)) return sel;
    }
    return null;
  }

  function build() {
    var sel = findSelect();
    if (!sel) return false;
    if (sel.getAttribute('data-tceb-stepper')) return true;
    sel.setAttribute('data-tceb-stepper', '1');
    var max = 0;
    for (var i = 0; i < sel.options.length; i++) max = Math.max(max, parseInt(sel.options[i].value, 10) || 0);

    var field = sel.closest('.has-float-label') || sel.parentElement;
    field.style.display = 'none';
    var row = document.createElement('div');
    row.className = 'tceb-team-row';
    // Counter first, label after it (mirrors the VIP checkbox + label)
    row.innerHTML = '<span class="tceb-stepper"><button type="button" aria-label="Fewer seats">&minus;</button>' +
      '<output aria-live="polite">0</output><button type="button" aria-label="More seats">+</button></span>' +
      '<span class="tceb-team-label">Extra team seats</span>';
    field.parentNode.insertBefore(row, field.nextSibling);
    var minus = row.querySelectorAll('button')[0], plus = row.querySelectorAll('button')[1], out = row.querySelector('output');

    function show() {
      var n = parseInt(sel.value, 10) || 0;
      out.textContent = n; minus.disabled = n <= 0; plus.disabled = n >= max;
    }
    function set(n) {
      n = Math.max(0, Math.min(max, n));
      sel.value = String(n);
      sel.dispatchEvent(new Event('change', { bubbles: true }));  // Spiffy reprices exactly as for a dropdown pick
      show();
    }
    minus.addEventListener('click', function () { set((parseInt(sel.value, 10) || 0) - 1); });
    plus.addEventListener('click', function () { set((parseInt(sel.value, 10) || 0) + 1); });
    sel.addEventListener('change', show);
    show();
    return true;
  }

  // Spiffy draws the form after load, and can redraw it: build once it appears, and again if it is redrawn
  tcebAfterMount(function () {
    var tries = 0;
    (function wait() { if (!build() && tries++ < 100) setTimeout(wait, 100); })();
    new MutationObserver(function () { var s = findSelect(); if (s && !s.getAttribute('data-tceb-stepper')) build(); })
      .observe(document.documentElement, { childList: true, subtree: true });
  });
})();

/* 3. VIP and Team sections. Finds the section holding the VIP checkbox and the one holding the Extra Team Seats
      dropdown and builds the rest of each box: red corner tab (the benefit or the saving, the biggest thing in the box), headline, grey line, bullets, price on the right
      (prepay prices on the -pif cart), and a "Get more details from Darren" link that plays Darren's video inline.
      Also styles the action rows (champagne tint, 44px checkbox), relabels the checkbox "Add VIP to my seat" and
      takes the apostrophe out of "FOUNDER'S" on the Join button, and hides the Rearden Sponsorship checkbox. If the script doesn't run, the plain Spiffy
      checkbox and dropdown show instead and still work. */
(function () {
  var PIF = /-pif\//.test(location.pathname + '/');   // the prepay cart's address ends in -pif
  var BOX = {
    vip: {
      tab: 'Extra half-day with Darren',
      price: PIF ? ['Only \u00241,495'] : ['Only \u0024115 today', '+\u0024115 for 12 months'],
      head: 'YES! Make me a VIP',
      sub: 'Private additional session · April 28, 2027',
      bullets: ['Connect with fellow VIP Founders before the event',
        'Your own VIP virtual space throughout the event',
        'Two weeks of recording access instead of 96 hours'],
      video: '1236615000'
    },
    team: {
      tab: 'Save \u00246,000 per seat',
      price: PIF ? ['Only \u00243,995/ea'] : ['Only \u0024308/ea today', '+\u0024308/ea for 12 months'],
      head: 'YES! I want to bring my leaders',
      sub: 'General Admission seats for your key leaders',
      bullets: ['They see the vision and share your conviction firsthand',
        'You come back with a team ready to lead the change',
        'Option to upgrade seat(s) to VIP before the event'],
      video: '1236615058'
    }
  };

  var font = document.createElement('link');
  font.rel = 'stylesheet';
  font.href = 'https://fonts.googleapis.com/css2?family=Bebas+Neue&display=swap';
  document.head.appendChild(font);

  var css = document.createElement('style');
  css.textContent =
    '.tceb-box .card{position:relative;overflow:hidden}' +
    '.tceb-tab{position:absolute;top:0;right:0;z-index:1;padding:12px 16px;background:#e1132e;color:#fff;font-size:17px;font-weight:800;letter-spacing:.08em;line-height:1.2;text-transform:uppercase}' +
    '.tceb-box .section__inner{padding-top:66px!important}' +
    '.tceb-content{position:relative;color:#000}' +
    '.tceb-head{margin:0 0 10px;padding-right:170px;font-family:"Bebas Neue",sans-serif;font-size:30px;font-weight:400;line-height:.95;letter-spacing:.02em;text-transform:uppercase}' +
    '.tceb-sub{margin:0 0 10px;padding-right:170px;color:#7a7a7a;font-size:13px;font-weight:600;letter-spacing:.04em;text-transform:uppercase}' +
    '.tceb-list{margin:0;padding-left:18px;font-size:14px;line-height:1.6}' +
    '.tceb-more{padding:0;border:0;background:none;color:inherit;font:inherit;text-decoration:underline;text-underline-offset:3px;cursor:pointer}' +
    '.tceb-price{position:absolute;top:-9px;right:0;text-align:right}' +
    '.tceb-price b{display:block;font-size:16px;font-weight:400;line-height:1.25}' +
    '.tceb-price small{display:block;margin-top:2px;color:#6e6e6e;font-size:12px}' +
    '.tceb-video{position:relative;margin-top:12px;padding-top:56.25%;background:#000}' +
    '.tceb-video iframe{position:absolute;top:0;left:0;width:100%;height:100%;border:0}' +
    '.tceb-close{position:absolute;top:8px;right:8px;z-index:2;display:flex;align-items:center;gap:6px;padding:6px 10px;border:0;border-radius:2px;background:rgba(0,0,0,.75);color:#fff;font-size:12px;font-weight:700;letter-spacing:.06em;text-transform:uppercase;cursor:pointer}' +
    '.tceb-vip .layout-checkboxes1,.tceb-team .tceb-team-row{margin-top:16px;padding:14px 16px;border-radius:2px;background:#f5eedf}' +
    '.tceb-vip .layout-checkboxes1{padding-top:15px;padding-bottom:15px;margin:16px 10px 20px;max-width:calc(100% - 20px)}' +
    '.tceb-vip .layout-checkboxes1 .form-group,.tceb-vip .layout-checkboxes1 .custom-control{margin-bottom:0!important}' +
    '.tceb-vip .custom-control{padding-left:0}' +
    '.tceb-vip .custom-control-label{position:relative;display:flex;align-items:center;min-height:44px;padding-left:58px;font-size:17px;font-weight:700;cursor:pointer}' +
    '.tceb-vip .custom-control-label::before,.tceb-vip .custom-control-label::after{top:0!important;left:0!important;width:44px!important;height:44px!important;margin-top:0!important}' +
    '.tceb-vip .custom-control-label::before{border:1px solid #dedede;border-radius:2px;background:#fff;box-shadow:none}' +
    '.tceb-vip .custom-control-input:checked~.custom-control-label::before{border-color:#e1132e!important;background-color:#e1132e!important}' +
    '.tceb-vip .custom-control-label::after{background-size:22px}' +
    '.tceb-vip .option-price{display:none!important}' +
    '.tceb-hide{display:none!important}' +
    '@media (max-width:600px){.tceb-box .section__inner{padding-top:60px!important}.tceb-head,.tceb-sub{padding-right:0}' +
    '.tceb-price{position:static;margin:0 0 12px}.tceb-vip .layout-checkboxes1{margin-bottom:13px}}';
  document.head.appendChild(css);

  function esc(t) { var d = document.createElement('div'); d.textContent = t; return d.innerHTML; }

  function build(sec, kind) {
    var c = BOX[kind], card = sec.querySelector('.card'), inner = sec.querySelector('.section__inner') || card;
    if (!card || card.querySelector('.tceb-tab')) return;
    var tab = document.createElement('span');
    tab.className = 'tceb-tab'; tab.textContent = c.tab;
    card.insertBefore(tab, card.firstChild);

    var box = document.createElement('div');
    box.className = 'tceb-content';
    box.innerHTML =
      '<div class="tceb-price"><b>' + esc(c.price[0]) + '</b>' + (c.price[1] ? '<small>' + esc(c.price[1]) + '</small>' : '') + '</div>' +
      '<h3 class="tceb-head">' + esc(c.head) + '</h3>' +
      '<p class="tceb-sub">' + esc(c.sub) + '</p>' +
      '<ul class="tceb-list">' + c.bullets.map(function (b) { return '<li>' + esc(b) + '</li>'; }).join('') +
      '<li><button type="button" class="tceb-more">Get more details from Darren →</button></li></ul>';
    inner.insertBefore(box, inner.firstChild);

    // Darren's video plays right under the bullets. Close it with the Close button on the video, or by clicking
    // the link again (which reads "Close the video" while it's open).
    var more = box.querySelector('.tceb-more'), LINK = more.textContent;
    function close() {
      var open = box.querySelector('.tceb-video');
      if (open) open.remove();
      more.textContent = LINK;
    }
    more.addEventListener('click', function () {
      if (box.querySelector('.tceb-video')) { close(); return; }
      var v = document.createElement('div');
      v.className = 'tceb-video';
      v.innerHTML = '<button type="button" class="tceb-close" aria-label="Close the video">Close <span aria-hidden="true">&times;</span></button>' +
        '<iframe src="https://player.vimeo.com/video/' + c.video + '?autoplay=1&title=0&byline=0&portrait=0&color=e02826&colors=000000,e02826,ffffff,000000&cc=0&pip=0&transcript=0&quality_selector=0&airplay=0&chromecast=0&speed=1" allow="autoplay; fullscreen; picture-in-picture" allowfullscreen title="Darren on the ' + (kind === 'vip' ? 'VIP upgrade' : 'team seats') + '"></iframe>';
      v.querySelector('.tceb-close').addEventListener('click', close);
      box.appendChild(v);
      more.textContent = 'Close the video \u00d7';
    });
  }

  function tag(block, kind) {
    var sec = block && block.closest('.section');
    if (!sec) return;
    if (!sec.classList.contains('tceb-' + kind)) sec.classList.add('tceb-box', 'tceb-' + kind);
    build(sec, kind);
  }

  function relabel(label) {
    // The checkbox label's own text (the price badge beside it is hidden by the CSS above)
    for (var n = label.firstChild; n; n = n.nextSibling) {
      if (n.nodeType === 3 && n.nodeValue.trim()) {
        if (n.nodeValue.trim() !== 'Add VIP to my seat') n.nodeValue = ' Add VIP to my seat ';
        return;
      }
    }
  }

  function scan() {
    document.querySelectorAll('[id^="block-"]').forEach(function (b) {
      var cbLabel = b.querySelector('input[type="checkbox"] ~ label');
      if (cbLabel && /vip/i.test(cbLabel.textContent)) { tag(b, 'vip'); relabel(cbLabel); }
      if (b.querySelector('select') && /extra team seats/i.test(b.textContent)) tag(b, 'team');
      // The Rearden Sponsorship is always included: hide its ticked checkbox so it can't be unticked
      // (it still shows in Order Details, see part 1)
      if (cbLabel && /rearden/i.test(cbLabel.textContent)) {
        var sec = b.closest('.section');
        var hide = sec && sec.querySelectorAll('[id^="block-"]').length === 1 ? sec : b;
        if (!hide.classList.contains('tceb-hide')) hide.classList.add('tceb-hide');
      }
    });
    // Join button: FOUNDER'S -> FOUNDERS
    document.querySelectorAll('button, .btn').forEach(function (btn) {
      for (var n = btn.firstChild; n; n = n.nextSibling) {
        if (n.nodeType === 3 && /FOUNDER['’]S/i.test(n.nodeValue)) n.nodeValue = n.nodeValue.replace(/(FOUNDER)['’](S)/gi, function (m, a, b) { return a + b; });
      }
    });
  }
  tcebAfterMount(function () {
    var tries = 0;
    (function wait() { scan(); if (!document.querySelector('.tceb-vip') && tries++ < 100) setTimeout(wait, 100); })();
    new MutationObserver(scan).observe(document.documentElement, { childList: true, subtree: true, characterData: true });
  });
})();
</script>
